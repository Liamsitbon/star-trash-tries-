cmake_minimum_required(VERSION 3.22)
project(QuestShaderProbe LANGUAGES CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "This is an Android EGL/GLES probe, not a PC compatibility simulator")
endif()
add_executable(quest_shader_probe main.cpp)
target_compile_features(quest_shader_probe PRIVATE cxx_std_17)
target_compile_options(quest_shader_probe PRIVATE -Wall -Wextra -Werror)
target_link_libraries(quest_shader_probe PRIVATE EGL GLESv3)
