cmake_minimum_required(VERSION 3.22)
project(SynapseQuestCore VERSION 0.1.0 LANGUAGES CXX)
option(SYNAPSE_CORE_TESTS "Build portable protocol tests" ON)
find_package(Threads REQUIRED)
add_library(SynapseQuestCore STATIC src/Protocol.cpp src/Models.cpp src/TcpTransport.cpp src/Session.cpp)
target_link_libraries(SynapseQuestCore PUBLIC Threads::Threads)
target_compile_features(SynapseQuestCore PUBLIC cxx_std_20)
target_include_directories(SynapseQuestCore PUBLIC include)
target_include_directories(SynapseQuestCore SYSTEM PRIVATE third_party)
target_compile_options(SynapseQuestCore PRIVATE -Wall -Wextra -Werror -pedantic)
if(SYNAPSE_CORE_TESTS)
  enable_testing()
  add_executable(protocol_test tests/protocol_test.cpp)
  target_link_libraries(protocol_test PRIVATE SynapseQuestCore)
  target_compile_options(protocol_test PRIVATE -UNDEBUG)
  add_test(NAME synapse_protocol COMMAND protocol_test)
  add_executable(models_test tests/models_test.cpp)
  target_link_libraries(models_test PRIVATE SynapseQuestCore)
  target_compile_options(models_test PRIVATE -UNDEBUG)
  add_test(NAME synapse_models COMMAND models_test)
  add_executable(transport_test tests/transport_test.cpp)
  target_link_libraries(transport_test PRIVATE SynapseQuestCore)
  target_compile_options(transport_test PRIVATE -UNDEBUG)
  add_test(NAME synapse_transport COMMAND transport_test)
  set_tests_properties(synapse_transport PROPERTIES TIMEOUT 25)
  add_executable(session_test tests/session_test.cpp)
  target_link_libraries(session_test PRIVATE SynapseQuestCore)
  target_compile_options(session_test PRIVATE -UNDEBUG)
  add_test(NAME synapse_session COMMAND session_test)
endif()
